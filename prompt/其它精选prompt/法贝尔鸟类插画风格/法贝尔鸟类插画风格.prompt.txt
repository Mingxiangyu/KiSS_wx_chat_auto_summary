## 自动提取信息
系统将自动从您提供的聊天记录中提取以下信息：
- 群名称：将从聊天记录的系统通知或常见群聊信息中提取
- 日期范围：根据记录中的所有日期自动生成（格式：YYYY-MM-DD ~ YYYY-MM-DD）
- 时间范围：根据记录中的首条和末条消息时间确定
- 昵称：昵称不需要显示账号

## 日报模式选择
- 日报模式：[完整版/简化版] (默认为完整版)
- 如果需要简化版，请在提交时注明"生成简化版日报"

## 简化版说明
如选择"简化版"，将只生成以下核心部分：
- 时段讨论热点（最多3个）
- 重要消息汇总
- 话唠榜（仅前3名）
- 简化版词云
日报内容更精简，适合快速浏览和分享。

## 聊天记录支持格式
支持以下多种常见格式：
- "[时间] 昵称：消息内容"
- "时间 - 昵称：消息内容"
- "昵称 时间：消息内容"
- 其他合理的时间和昵称分隔格式

如未能识别消息格式或未找到有效记录，将显示提示信息并尝试按最佳猜测处理。

    function makeDraggable(containerElement) {
        let isDragging = false;
        let startX, startY, scrollLeftStart, scrollTopStart;

        if (!containerElement) return;

        containerElement.addEventListener('mousedown', (e) => {
            if (e.button !== 0) return;
            if (e.target.closest('button') || e.target.closest('a') || e.target.closest('[onclick]')) return;

            if (e.target === containerElement) {
                const verticalScrollbarVisible = containerElement.scrollHeight > containerElement.clientHeight;
                const horizontalScrollbarVisible = containerElement.scrollWidth > containerElement.clientWidth;
                const clickedOnVerticalScrollbar = verticalScrollbarVisible && e.offsetX >= containerElement.clientWidth - 17;
                const clickedOnHorizontalScrollbar = horizontalScrollbarVisible && e.offsetY >= containerElement.clientHeight - 17;
                if (clickedOnVerticalScrollbar || clickedOnHorizontalScrollbar) return;
            }

            isDragging = true;
            startX = e.pageX; startY = e.pageY;
            scrollLeftStart = containerElement.scrollLeft; scrollTopStart = containerElement.scrollTop;
            containerElement.classList.add('dragging');
        });

        document.addEventListener('mousemove', (e) => {
            if (!isDragging) return;
            e.preventDefault();
            const deltaX = e.pageX - startX; const deltaY = e.pageY - startY;
            containerElement.scrollLeft = scrollLeftStart - deltaX;
            containerElement.scrollTop = scrollTopStart - deltaY;
        });

        document.addEventListener('mouseup', (e) => {
            if (e.button !== 0 || !isDragging) return;
            isDragging = false;
            containerElement.classList.remove('dragging');
        });

        containerElement.addEventListener('touchstart', (e) => {
            if (e.target.closest('button') || e.target.closest('a') || e.target.closest('[onclick]')) return;
            if (e.touches.length === 1) {
                isDragging = true;
                const touch = e.touches[0];
                startX = touch.pageX;
                startY = touch.pageY;
                scrollLeftStart = containerElement.scrollLeft;
                scrollTopStart = containerElement.scrollTop;
                containerElement.classList.add('dragging');
            }
        }, { passive: true });

        document.addEventListener('touchmove', (e) => {
            if (!isDragging || e.touches.length !== 1) return;
            const touch = e.touches[0];
            const deltaX = touch.pageX - startX;
            const deltaY = touch.pageY - startY;
            containerElement.scrollLeft = scrollLeftStart - deltaX;
            containerElement.scrollTop = scrollTopStart - deltaY;
        });

        document.addEventListener('touchend', (e) => {
            if (!isDragging) return;
            isDragging = false;
            containerElement.classList.remove('dragging');
        });
    }

    async function downloadMermaidAsSVG(svgElement, baseFilename, feedbackElem) {
        if (svgElement && svgElement.innerHTML.trim() !== "") {
            try {
                const svgData = new XMLSerializer().serializeToString(svgElement);
                const blob = new Blob([svgData], { type: 'image/svg+xml;charset=utf-8' });
                const url = URL.createObjectURL(blob);
                const link = document.createElement('a');
                link.href = url;
                link.download = `${baseFilename}.svg`;
                document.body.appendChild(link);
                link.click();
                document.body.removeChild(link);
                URL.revokeObjectURL(url);
                showMermaidFeedback(feedbackElem, `${baseFilename}.svg 已下载!`, 'success');
            } catch (error) {
                console.error(`Error during SVG download for ${baseFilename}:`, error);
                showMermaidFeedback(feedbackElem, `下载 ${baseFilename}.svg 失败: ${error.message}`, 'error');
            }
        } else {
            showMermaidFeedback(feedbackElem, '无法下载：图表内容为空。', 'error');
        }
    }

            const {svg, bindFunctions} = await mermaid.render(uniqueSvgId, definition);
            targetContainer.innerHTML = svg;

            const svgElement = targetContainer.querySelector('svg');
            if (svgElement) {
                if (targetContainer === mainMermaidContainer) mainCurrentScale = 1;
                else if (targetContainer === modalMermaidContainer) modalCurrentScale = 1;

                makeDraggable(targetContainer);

                targetContainer.innerHTML = '';
                targetContainer.classList.add('empty');

    async function captureScreenshot(elementToCapture) {
        // 检查html2canvas是否可用
        if (typeof html2canvas === 'undefined') {
            throw new Error('html2canvas库未正确加载，请刷新页面重试');
        }

        try {
            await document.fonts.ready;
            console.log('开始html2canvas截图...');

            // 简化且稳定的配置
            const canvas = await html2canvas(elementToCapture, {
                useCORS: true,
                scale: window.devicePixelRatio || 1,
                logging: false,
                backgroundColor: '#f2e8c9',
                allowTaint: false,
                foreignObjectRendering: false,
                imageTimeout: 15000,
                removeContainer: true,
                scrollX: 0,
                scrollY: 0,
                // 过滤不需要截图的交互元素
                ignoreElements: function(element) {
                    return element.classList.contains('action-buttons-container') ||
                           element.id === 'feedback-message' ||
                           element.classList.contains('modal-overlay');
                },
                onclone: function(clonedDoc, element) {
                    try {
                        // 隐藏交互元素
                        const actionButtons = clonedDoc.querySelector('.action-buttons-container');
                        const feedbackMsg = clonedDoc.querySelector('#feedback-message');
                        const modalOverlay = clonedDoc.querySelector('.modal-overlay');

                        if (actionButtons) actionButtons.style.display = 'none';
                        if (feedbackMsg) feedbackMsg.style.display = 'none';
                        if (modalOverlay) modalOverlay.style.display = 'none';

                        // 确保body样式正确
                        const clonedBody = clonedDoc.body;
                        if (clonedBody) {
                            clonedBody.style.transform = 'none';
                            clonedBody.style.overflow = 'visible';
                            clonedBody.style.position = 'static';
                        }
                    } catch (cloneError) {
                        console.warn('Clone处理出现警告:', cloneError);
                        // 不抛出错误，继续截图过程
                    }
                }
            });

            console.log('截图成功！');
            return canvas;

        } catch (error) {
            console.error('html2canvas错误详情:', error);

            // 如果是常见的html2canvas错误，尝试简化的备用方案
            if (error.message.includes('Unable to find element') ||
                error.message.includes('createPattern') ||
                error.message.includes('canvas element with a width or height of 0') ||
                error.message.includes('cloned iframe')) {
                console.log('检测到html2canvas兼容性问题，尝试简化的备用方案...');

                try {
                    // 简化的备用方案：使用最基本的配置重试
                    const backupCanvas = await html2canvas(elementToCapture, {
                        scale: 1,
                        logging: false,
                        backgroundColor: '#f2e8c9',
                        allowTaint: true,
                        useCORS: false,
                        foreignObjectRendering: false,
                        imageTimeout: 0,
                        removeContainer: true,
                        ignoreElements: function(element) {
                            // 忽略所有可能有问题的元素
                            return element.classList.contains('action-buttons-container') ||
                                   element.id === 'feedback-message' ||
                                   element.classList.contains('modal-overlay') ||
                                   element.tagName === 'SCRIPT' ||
                                   element.tagName === 'STYLE';
                        }
                    });

                    console.log('简化备用方案成功！');
                    return backupCanvas;

                } catch (backupError) {
                    console.error('备用方案也失败了:', backupError);

                    // 提供用户友好的错误信息和解决建议
                    throw new Error(`截图功能暂时不可用。建议解决方案：
1. 刷新页面后重试
2. 使用浏览器的打印功能（Ctrl+P）保存为PDF
3. 使用浏览器的开发者工具手动截图

技术详情：${backupError.message}`);
                }
            }

            // 提供更具体的错误信息
            if (error.message.includes('SecurityError')) {
                throw new Error('截图失败：页面包含跨域内容，请确保所有资源都来自同一域名');
            } else if (error.message.includes('timeout')) {
                throw new Error('截图失败：处理超时，页面内容过于复杂');
            } else {
                throw new Error(`截图失败：${error.message || '未知错误，请尝试刷新页面后重试'}`);
            }
        }
    }

            }

            // 使用Promise包装toBlob以便更好地处理错误
            const blob = await new Promise((resolve, reject) => {
                canvas.toBlob((blob) => {
                    if (blob) {
                        resolve(blob);
                    } else {
                        reject(new Error('无法将Canvas转换为Blob'));
                    }
                }, 'image/png', 0.95);
            });

            if (navigator.clipboard && navigator.clipboard.write) {
                await navigator.clipboard.write([
                    new ClipboardItem({ 'image/png': blob })
                ]);
                showReportFeedbackMessage('✅ 截图已拷贝到剪贴板!', 'success');
            } else {
                showReportFeedbackMessage('截图已生成，但您的浏览器不支持直接拷贝图片。请尝试下载截图。', 'error', 5000);
            }

        } catch (error) {
            console.error('截图操作失败:', error);
            showReportFeedbackMessage(`截图失败: ${error.message}`, 'error', 5000);
        } finally {
            this.innerHTML = originalButtonText;
            this.disabled = false;
        }
    });

    const screenshotDownloadBtn = document.getElementById('screenshotDownloadBtn');
    if(screenshotDownloadBtn) screenshotDownloadBtn.addEventListener('click', async function() {
        const reportContent = document.body;
        if (!reportContent) {
            showReportFeedbackMessage('截图错误: 未找到报告内容。', 'error');
            return;
        }

        const originalButtonText = this.innerHTML;
        this.innerHTML = '⏳ 生成中...';
        this.disabled = true;

        try {
            showReportFeedbackMessage('正在生成截图，请稍候...', 'info');

            const canvas = await captureScreenshot(reportContent);

            if (!canvas) {
                throw new Error('截图生成失败，canvas为空');
            }

            // 使用toDataURL生成图片数据
            const imageData = canvas.toDataURL('image/png', 0.95);

            if (!imageData || imageData === 'data:,') {
                throw new Error('无法生成图片数据');
            }

            // 创建下载链接
            const link = document.createElement('a');
            const webpageName = getWebpageNameForFilename();
            const dateStr = getDateStringForFilename();
            const filename = sanitizeFilename(`${webpageName}_${dateStr}_截图.png`);

            link.href = imageData;
            link.download = filename;
            link.style.display = 'none';

            // 添加到DOM，触发下载，然后移除
            document.body.appendChild(link);
            link.click();

            // 延迟移除链接，确保下载开始
            setTimeout(() => {
                document.body.removeChild(link);
            }, 100);

            showReportFeedbackMessage(`✅ 截图已下载: ${filename}`, 'success');

        } catch (error) {
            console.error('截图下载失败:', error);
            showReportFeedbackMessage(`截图下载失败: ${error.message}`, 'error', 5000);
        } finally {
            this.innerHTML = originalButtonText;
            this.disabled = false;
        }
    });

    const copyHtmlBtn = document.getElementById('copyHtmlBtn');
    if(copyHtmlBtn) copyHtmlBtn.addEventListener('click', async function() {
        try {
            const htmlContent = document.documentElement.outerHTML;
            await navigator.clipboard.writeText(htmlContent);
            showReportFeedbackMessage('HTML源码已拷贝到剪贴板!', 'success');
        } catch (error) {
            console.error('拷贝HTML失败:', error);
            showReportFeedbackMessage('拷贝HTML失败: ' + error.message, 'error');
        }
    });

    const downloadHtmlBtn = document.getElementById('downloadHtmlBtn');
    if(downloadHtmlBtn) downloadHtmlBtn.addEventListener('click', function() {
        try {
            const htmlContent = document.documentElement.outerHTML;
            const blob = new Blob([htmlContent], { type: 'text/html;charset=utf-8' });
            const url = URL.createObjectURL(blob);
            const link = document.createElement('a');

            const webpageName = getWebpageNameForFilename();
            const dateStr = getDateStringForFilename();
            const filename = sanitizeFilename(`${webpageName}_${dateStr}_报告.html`);

            link.href = url;
            link.download = filename;
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
            URL.revokeObjectURL(url);

            showReportFeedbackMessage(`HTML报告已下载: ${filename}`, 'success');
        } catch (error) {
            console.error('下载HTML失败:', error);
            showReportFeedbackMessage('下载HTML失败: ' + error.message, 'error');
        }
    });

    // 页面加载完成后初始化
    document.addEventListener('DOMContentLoaded', function() {
        // 检查依赖库是否加载
        if (typeof mermaid === 'undefined') {
            showReportFeedbackMessage('Mermaid库未正确加载，思维导图功能可能无法使用', 'error', 5000);
        }
        if (typeof html2canvas === 'undefined') {
            showReportFeedbackMessage('html2canvas库未正确加载，截图功能可能无法使用', 'error', 5000);
        }

        // 初始化思维导图
        initializeMainDiagram();
    });
    </script>

</body></html>
